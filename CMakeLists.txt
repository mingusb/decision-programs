cmake_minimum_required(VERSION 3.30)
if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
  set(CMAKE_CUDA_ARCHITECTURES 86-real CACHE STRING "CUDA architectures to build")
endif()
project(gpu_histogram VERSION 0.1.0 LANGUAGES CXX CUDA)
find_package(CUDAToolkit 13.4 REQUIRED)
# CMake 4.2's NVIDIA table stops at CUDA C++20 although nvcc 13.4 supports C++23.
# Keep the actual dialect explicit instead of silently lowering the language mode.
if(CMAKE_CUDA_COMPILER_ID STREQUAL "NVIDIA" AND CMAKE_CUDA_COMPILER_VERSION VERSION_GREATER_EQUAL 13.4)
  set(CMAKE_CUDA23_STANDARD_COMPILE_OPTION "-std=c++23")
  set(CMAKE_CUDA23_EXTENSION_COMPILE_OPTION "-std=c++23")
endif()
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CUDA_STANDARD 23)
set(CMAKE_CUDA_STANDARD_REQUIRED ON)
include(cmake/CompilerDiagnostics.cmake)
gh_configure_cuda_diagnostics()
option(GH_PROFILE_HOST_SYMBOLS "Add host symbols/frame pointers for diagnostic profiling; retain optimized device code" OFF)
option(GH_DEVICE_DEBUG "Build unoptimized CUDA device debug code (-G); correctness debugging only, never performance evidence" OFF)
if(GH_DEVICE_DEBUG)
  set(GH_CUDA_OPTIMIZATION_FLAGS "")
  add_compile_options("$<$<COMPILE_LANGUAGE:CXX>:-g;-O0>" "$<$<COMPILE_LANGUAGE:CUDA>:-g;-G;-O0>")
  message(STATUS "GH_DEVICE_DEBUG: unoptimized correctness-debug build; do not use timings for performance rankings")
else()
  set(GH_CUDA_OPTIMIZATION_FLAGS -O3 -lineinfo)
endif()
if(GH_PROFILE_HOST_SYMBOLS)
  add_compile_options("$<$<COMPILE_LANGUAGE:CXX>:-g;-fno-omit-frame-pointer>"
                      "$<$<COMPILE_LANGUAGE:CUDA>:-g;-Xcompiler=-fno-omit-frame-pointer>")
  if(NOT GH_DEVICE_DEBUG)
    add_compile_options($<$<COMPILE_LANGUAGE:CUDA>:-lineinfo>)
  endif()
endif()
# Policy selection and structural validation use CUDA types, but no runtime API.
# Keep this target host-only so its tests neither link nor load the CUDA runtime.
add_library(gh_policy STATIC src/defaults.cpp src/config.cpp)
target_include_directories(gh_policy PUBLIC include)
target_include_directories(gh_policy SYSTEM PUBLIC ${CUDAToolkit_INCLUDE_DIRS})
add_library(gh STATIC src/histogram.cu src/bitplane.cu src/global_narrow.cu src/shared_overflow.cu src/global_window.cu)
target_include_directories(gh PUBLIC include)
target_link_libraries(gh PUBLIC gh_policy CUDA::cudart)
target_compile_options(gh PRIVATE "$<$<COMPILE_LANGUAGE:CUDA>:${GH_CUDA_OPTIMIZATION_FLAGS};--ptxas-options=-v>")
option(GH_BUILD_BENCHMARKS "Build benchmarks and their isolated NVIDIA reference implementations" ON)
if(GH_BUILD_BENCHMARKS)
  add_library(gh_benchmark_references STATIC bench/references.cu bench/sample256.cu)
  target_include_directories(gh_benchmark_references PRIVATE include)
  target_link_libraries(gh_benchmark_references PUBLIC CUDA::cudart)
  target_compile_options(gh_benchmark_references PRIVATE "$<$<COMPILE_LANGUAGE:CUDA>:${GH_CUDA_OPTIMIZATION_FLAGS};--ptxas-options=-v>")
  add_executable(histogram_bench bench/benchmark.cpp bench/cache_flush.cu)
  target_compile_options(histogram_bench PRIVATE "$<$<COMPILE_LANGUAGE:CUDA>:${GH_CUDA_OPTIMIZATION_FLAGS}>")
  target_link_libraries(histogram_bench PRIVATE gh gh_benchmark_references ${CMAKE_DL_LIBS})
  target_include_directories(histogram_bench PRIVATE support)
  add_executable(histogram_cache_probe bench/cache_probe.cpp bench/cache_flush.cu)
  target_compile_options(histogram_cache_probe PRIVATE "$<$<COMPILE_LANGUAGE:CUDA>:${GH_CUDA_OPTIMIZATION_FLAGS}>")
  target_link_libraries(histogram_cache_probe PRIVATE gh ${CMAKE_DL_LIBS})
  target_include_directories(histogram_cache_probe PRIVATE support)
endif()
add_executable(histogram_correctness tests/correctness.cpp)
target_link_libraries(histogram_correctness PRIVATE gh)
target_include_directories(histogram_correctness PRIVATE support)
add_executable(histogram_defaults tests/defaults.cpp)
target_link_libraries(histogram_defaults PRIVATE gh_policy)
add_executable(histogram_shared_overflow_tests tests/shared_overflow.cpp)
target_link_libraries(histogram_shared_overflow_tests PRIVATE gh)
target_include_directories(histogram_shared_overflow_tests PRIVATE support)
add_executable(histogram_global_window_tests tests/global_window.cpp)
target_link_libraries(histogram_global_window_tests PRIVATE gh)
target_include_directories(histogram_global_window_tests PRIVATE support)
add_executable(histogram_skew_generation tests/skew_generation.cpp)
target_link_libraries(histogram_skew_generation PRIVATE gh_policy)
target_include_directories(histogram_skew_generation PRIVATE support)
enable_testing()
gh_add_gpu_test(NAME histogram_correctness COMMAND histogram_correctness)
add_test(NAME histogram_defaults COMMAND histogram_defaults)
gh_add_gpu_test(NAME histogram_shared_overflow COMMAND histogram_shared_overflow_tests)
gh_add_gpu_test(NAME histogram_global_window COMMAND histogram_global_window_tests)
add_test(NAME histogram_skew_generation COMMAND histogram_skew_generation)
set_tests_properties(histogram_shared_overflow PROPERTIES SKIP_RETURN_CODE 77)
find_package(Python3 COMPONENTS Interpreter)
if(Python3_Interpreter_FOUND)
  add_test(NAME autotune_policy COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_autotune.py)
endif()
