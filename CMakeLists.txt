add_library(gh_data_leaf OBJECT gpu_histogram.cu)
target_compile_definitions(gh_data_leaf PRIVATE GH_IMPLEMENTATION=0 GH_DATA_LEAF_IMPLEMENTATION=1 GH_MODE=0 GH_OBSERVE=$<BOOL:${GH_OBSERVE}>)
target_link_libraries(gh_data_leaf PRIVATE CUDA::cudart)
set_target_properties(gh_data_leaf PROPERTIES CUDA_SEPARABLE_COMPILATION ON)

add_library(gh STATIC gpu_histogram.cu $<TARGET_OBJECTS:gh_data_leaf>)
target_compile_definitions(gh PRIVATE GH_IMPLEMENTATION=1 GH_MODE=0)
target_compile_definitions(gh PUBLIC GH_OBSERVE=$<BOOL:${GH_OBSERVE}>)
target_link_libraries(gh PUBLIC CUDA::cudart)
set_target_properties(gh PROPERTIES CUDA_SEPARABLE_COMPILATION ON)
target_compile_options(gh PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--ptxas-options=-v;-Xcompiler=-Wall,-Wextra>)
if(GH_OBSERVE)
  target_compile_options(gh PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-lineinfo>)
  target_compile_options(gh_data_leaf PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-lineinfo>)
endif()
enable_testing()
function(gh_executable name mode)
  add_executable(${name} gpu_histogram.cpp gpu_histogram.cu)
  target_compile_definitions(${name} PRIVATE GH_MODE=${mode} GH_IMPLEMENTATION=0)
  if(mode EQUAL 13)
    target_sources(${name} PRIVATE $<TARGET_OBJECTS:gh_data_leaf>)
    target_link_libraries(${name} PRIVATE CUDA::cudart)
    target_compile_definitions(${name} PRIVATE GH_OBSERVE=$<BOOL:${GH_OBSERVE}>)
  else()
    target_link_libraries(${name} PRIVATE gh)
  endif()
  set_target_properties(${name} PROPERTIES CUDA_SEPARABLE_COMPILATION ON)
  target_compile_options(${name} PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-UNDEBUG>)
endfunction()
function(gh_check family mode)
  gh_executable(${family}_checks ${mode})
  add_test(NAME ${family}_checks COMMAND ${family}_checks)
  set_tests_properties(${family}_checks PROPERTIES LABELS gpu RUN_SERIAL TRUE)
endfunction()
