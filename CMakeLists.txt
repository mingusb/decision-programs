cmake_minimum_required(VERSION 3.30)
project(gpu_histogram LANGUAGES CXX CUDA)
if(CMAKE_CUDA_COMPILER_ID STREQUAL "NVIDIA" AND CMAKE_CUDA_COMPILER_VERSION VERSION_GREATER_EQUAL 13.4)
  set(CMAKE_CUDA23_STANDARD_COMPILE_OPTION "-std=c++23")
  set(CMAKE_CUDA23_EXTENSION_COMPILE_OPTION "-std=c++23")
endif()
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_CUDA_STANDARD 23)
set(CMAKE_CUDA_STANDARD_REQUIRED ON)
set(CMAKE_CUDA_EXTENSIONS OFF)
find_package(CUDAToolkit REQUIRED)
option(GH_OBSERVE "Compile GPU stage markers for diagnostic builds" OFF)
add_library(gh STATIC src/core.cu src/learn.cu src/metrics.cu src/observe.cu src/count.cu src/data.cu src/model.cu src/prediction.cu src/csv.cu src/reference.cu)
target_compile_definitions(gh PUBLIC GH_OBSERVE=$<BOOL:${GH_OBSERVE}>)
target_include_directories(gh PUBLIC include)
target_link_libraries(gh PUBLIC CUDA::cudart)
set_target_properties(gh PROPERTIES CUDA_SEPARABLE_COMPILATION ON)
target_compile_options(gh PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--ptxas-options=-v;-Xcompiler=-Wall,-Wextra>)
if(GH_OBSERVE)
  target_compile_options(gh PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-lineinfo>)
endif()
enable_testing()
function(gh_check family)
  add_executable(${family}_checks tests/driver.cu tests/${family}_checks.cu)
  target_include_directories(${family}_checks PRIVATE tests)
  target_link_libraries(${family}_checks PRIVATE gh)
  set_target_properties(${family}_checks PROPERTIES CUDA_SEPARABLE_COMPILATION ON)
  target_compile_options(${family}_checks PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-UNDEBUG>)
  add_test(NAME ${family}_checks COMMAND ${family}_checks)
  set_tests_properties(${family}_checks PROPERTIES LABELS gpu RUN_SERIAL TRUE)
endfunction()
gh_check(core)
gh_check(count)
target_compile_definitions(count_checks PRIVATE GH_COUNT_RUNTIME=1)
gh_check(data)
gh_check(model)
gh_check(learn)
gh_check(observe)
gh_check(metrics)
gh_check(prediction)
gh_check(csv)
gh_check(reference)
