cmake_minimum_required(VERSION 3.30)
project(ghb_instrumentation VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
find_package(CUDAToolkit 13.4 REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
option(GHB_WITH_NVTX "Enable optional Nsight annotations" ON)
option(GHB_BUILD_HISTOGRAM_PROBE "Instrument existing, separately built count kernels" ON)
option(GHB_BUILD_TRAINER "Build the custom GPU tree trainer" ON)
option(GHB_BUILD_SORT_REFERENCE "Build isolated benchmark-only NVIDIA sorting comparisons" OFF)
option(GHB_BUILD_DIAGNOSTIC_VALIDATION "Build optional CUDA stage/lifecycle validation executables" OFF)
option(GHB_BUILD_HARDWARE_CALIBRATION "Build optional standalone hardware calibration benchmark" OFF)
include("${CMAKE_CURRENT_SOURCE_DIR}/../cmake/CompilerDiagnostics.cmake")
if(NOT GHB_BUILD_TRAINER AND (GH_COMPILER_DIAGNOSTICS OR GH_COMPILER_TIME_TRACE OR GH_DEVICE_SANITIZE))
  message(FATAL_ERROR "GH CUDA diagnostic options require GHB_BUILD_TRAINER; imported count archives cannot be instrumented by this project")
endif()
option(GH_PROFILE_HOST_SYMBOLS "Add host symbols/frame pointers for diagnostic profiling; retain optimized device code" OFF)
option(GH_DEVICE_DEBUG "Build unoptimized CUDA device debug code (-G); correctness debugging only, never performance evidence" OFF)
if(GH_DEVICE_DEBUG)
  set(GH_CUDA_OPTIMIZATION_FLAGS "")
  add_compile_options("$<$<COMPILE_LANGUAGE:CXX>:-g;-O0>" "$<$<COMPILE_LANGUAGE:CUDA>:-g;-G;-O0>")
  message(STATUS "GH_DEVICE_DEBUG: unoptimized correctness-debug build; do not use timings for performance rankings")
else()
  set(GH_CUDA_OPTIMIZATION_FLAGS -O3 -lineinfo)
endif()
if(GH_PROFILE_HOST_SYMBOLS)
  add_compile_options("$<$<COMPILE_LANGUAGE:CXX>:-g;-fno-omit-frame-pointer>"
                      "$<$<COMPILE_LANGUAGE:CUDA>:-g;-Xcompiler=-fno-omit-frame-pointer>")
  if(NOT GH_DEVICE_DEBUG)
    add_compile_options($<$<COMPILE_LANGUAGE:CUDA>:-lineinfo>)
  endif()
endif()

add_library(ghb_instrumentation STATIC src/instrumentation.cpp)
target_include_directories(ghb_instrumentation PUBLIC include)
target_link_libraries(ghb_instrumentation PUBLIC CUDA::cudart PRIVATE ${CMAKE_DL_LIBS})
target_compile_options(ghb_instrumentation PRIVATE -Wall -Wextra -Wpedantic)
if(GHB_WITH_NVTX)
  target_compile_definitions(ghb_instrumentation PRIVATE GHB_WITH_NVTX=1)
endif()

enable_testing()
add_executable(ghb_instrumentation_tests tests/instrumentation.cpp)
target_link_libraries(ghb_instrumentation_tests PRIVATE ghb_instrumentation)
gh_add_gpu_test(NAME instrumentation_gpu COMMAND ghb_instrumentation_tests)
set_tests_properties(instrumentation_gpu PROPERTIES LABELS "gpu" RUN_SERIAL TRUE SKIP_RETURN_CODE 77)

add_executable(ghb_disabled_tests tests/disabled.cpp)
target_include_directories(ghb_disabled_tests PRIVATE include)
target_include_directories(ghb_disabled_tests SYSTEM PRIVATE ${CUDAToolkit_INCLUDE_DIRS})
# The disabled path deliberately has no CUDA or instrumentation linkage.
add_test(NAME instrumentation_disabled COMMAND ghb_disabled_tests)
set_tests_properties(instrumentation_disabled PROPERTIES LABELS "cpu")
add_executable(ghb_prediction_layout_tests tests/prediction_layout.cpp)
target_include_directories(ghb_prediction_layout_tests PRIVATE include)
target_compile_options(ghb_prediction_layout_tests PRIVATE -Wall -Wextra -Wpedantic)
# This size-arithmetic test has no CUDA/instrumentation linkage or device calls.
add_test(NAME prediction_layout COMMAND ghb_prediction_layout_tests)
set_tests_properties(prediction_layout PROPERTIES LABELS "cpu")
add_executable(ghb_finite_validation_tests tests/finite_validation.cpp)
target_include_directories(ghb_finite_validation_tests PRIVATE include)
target_compile_options(ghb_finite_validation_tests PRIVATE -Wall -Wextra -Wpedantic)
# Predicate and launch-plan arithmetic only: no CUDA linkage or device calls.
add_test(NAME prediction_finite_values COMMAND ghb_finite_validation_tests)
set_tests_properties(prediction_finite_values PROPERTIES LABELS "cpu")
foreach(test IN ITEMS observe evaluate)
  add_test(NAME instrumentation_${test}
    COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_${test}.py)
  set_tests_properties(instrumentation_${test} PROPERTIES LABELS "cpu")
endforeach()

if(GHB_BUILD_TRAINER)
  if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
    set(CMAKE_CUDA_ARCHITECTURES 86-real CACHE STRING "CUDA architectures to build")
  endif()
  enable_language(CUDA)
  if(CMAKE_CUDA_COMPILER_ID STREQUAL "NVIDIA" AND CMAKE_CUDA_COMPILER_VERSION VERSION_GREATER_EQUAL 13.4)
    set(CMAKE_CUDA23_STANDARD_COMPILE_OPTION "-std=c++23")
    set(CMAKE_CUDA23_EXTENSION_COMPILE_OPTION "-std=c++23")
  endif()
  set(CMAKE_CUDA_STANDARD 23)
  set(CMAKE_CUDA_STANDARD_REQUIRED ON)
  gh_configure_cuda_diagnostics()
  add_library(ghb STATIC src/booster.cpp src/kernels.cu src/initialization.cu src/quantize.cu src/resident.cu src/root_histogram.cu src/split_search.cu src/deeper_histogram.cu src/batch_resident.cu src/higher_order.cu src/prediction.cu src/finite_validation.cu)
  target_include_directories(ghb PUBLIC include)
  target_link_libraries(ghb PUBLIC ghb_instrumentation)
  target_compile_options(ghb PRIVATE
    $<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra;-Wpedantic>
    "$<$<COMPILE_LANGUAGE:CUDA>:${GH_CUDA_OPTIMIZATION_FLAGS}>")
  add_executable(ghb_higher_order_math_tests tests/higher_order_math.cpp)
  target_include_directories(ghb_higher_order_math_tests PRIVATE include)
  target_include_directories(ghb_higher_order_math_tests SYSTEM PRIVATE ${CUDAToolkit_INCLUDE_DIRS})
  target_compile_options(ghb_higher_order_math_tests PRIVATE -Wall -Wextra -Wpedantic)
  add_test(NAME trainer_higher_order_math COMMAND ghb_higher_order_math_tests)
  set_tests_properties(trainer_higher_order_math PROPERTIES LABELS "cpu")
  add_executable(ghb_higher_order_gpu_tests tests/higher_order_gpu.cu)
  target_link_libraries(ghb_higher_order_gpu_tests PRIVATE ghb)
  gh_add_gpu_test(NAME trainer_higher_order_gpu COMMAND ghb_higher_order_gpu_tests)
  set_tests_properties(trainer_higher_order_gpu PROPERTIES LABELS "gpu" RUN_SERIAL TRUE SKIP_RETURN_CODE 77)
  add_executable(ghb_bench bench/booster.cpp)
  target_link_libraries(ghb_bench PRIVATE ghb)
  target_include_directories(ghb_bench PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../support")
  target_compile_options(ghb_bench PRIVATE -Wall -Wextra -Wpedantic)
  add_executable(ghb_real_bench bench/real_data.cpp)
  target_link_libraries(ghb_real_bench PRIVATE ghb)
  target_include_directories(ghb_real_bench PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../support")
  add_executable(ghb_prediction_bench bench/prediction.cpp)
  target_link_libraries(ghb_prediction_bench PRIVATE ghb)
  target_include_directories(ghb_prediction_bench PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../support")
  add_executable(ghb_deeper_histogram_bench bench/deeper_histogram.cu)
  target_link_libraries(ghb_deeper_histogram_bench PRIVATE ghb)
  if(GHB_BUILD_DIAGNOSTIC_VALIDATION)
    add_executable(ghb_validate_gpu profiling/validate_gpu.cu)
    target_link_libraries(ghb_validate_gpu PRIVATE ghb)
  endif()
  if(GHB_BUILD_HARDWARE_CALIBRATION)
    add_executable(ghb_hardware_calibration profiling/calibrate.cu)
    target_link_libraries(ghb_hardware_calibration PRIVATE CUDA::cudart)
    target_compile_options(ghb_hardware_calibration PRIVATE "$<$<COMPILE_LANGUAGE:CUDA>:${GH_CUDA_OPTIMIZATION_FLAGS}>")
  endif()
  if(GHB_BUILD_SORT_REFERENCE)
    add_executable(ghb_sort_reference bench/sort_reference.cu)
    target_link_libraries(ghb_sort_reference PRIVATE CUDA::cudart)
    target_compile_options(ghb_sort_reference PRIVATE ${GH_CUDA_OPTIMIZATION_FLAGS})
  endif()
  foreach(test IN ITEMS booster kernels initialization quantize root_histogram split_search deeper_histogram batch_resident batch_training prediction)
    if(test STREQUAL "kernels" OR test STREQUAL "initialization" OR test STREQUAL "root_histogram" OR test STREQUAL "split_search" OR test STREQUAL "deeper_histogram" OR test STREQUAL "batch_resident")
      add_executable(ghb_${test}_tests tests/${test}.cu)
    else()
      add_executable(ghb_${test}_tests tests/${test}.cpp)
    endif()
    target_link_libraries(ghb_${test}_tests PRIVATE ghb)
    gh_add_gpu_test(NAME trainer_${test} COMMAND ghb_${test}_tests)
    set_tests_properties(trainer_${test} PROPERTIES LABELS "gpu" RUN_SERIAL TRUE SKIP_RETURN_CODE 77)
  endforeach()
endif()

if(GHB_BUILD_HISTOGRAM_PROBE)
  if(GH_COMPILER_DIAGNOSTICS OR GH_COMPILER_TIME_TRACE OR GH_DEVICE_SANITIZE)
    message(STATUS "Imported count libraries retain their original flags; configure the root project separately for count compiler/sanitizer coverage")
  endif()
  get_filename_component(GH_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
  set(GHB_COUNT_BUILD "${GH_ROOT}/build/window-experiment" CACHE PATH
      "Existing count libraries; this project never rebuilds them")
  foreach(lib IN ITEMS gh gh_policy)
    if(NOT EXISTS "${GHB_COUNT_BUILD}/lib${lib}.a")
      message(FATAL_ERROR "Missing ${GHB_COUNT_BUILD}/lib${lib}.a. Point GHB_COUNT_BUILD at an existing count build or disable GHB_BUILD_HISTOGRAM_PROBE.")
    endif()
    add_library(ghb_existing_${lib} STATIC IMPORTED)
    set_target_properties(ghb_existing_${lib} PROPERTIES IMPORTED_LOCATION "${GHB_COUNT_BUILD}/lib${lib}.a")
  endforeach()
  set(fingerprint "${CMAKE_CXX_COMPILER_ID}:${CMAKE_CXX_COMPILER_VERSION}:${CMAKE_CXX_COMPILER}:${GHB_WITH_NVTX}:${GH_PROFILE_HOST_SYMBOLS}:${GH_DEVICE_DEBUG}:${GH_COMPILER_DIAGNOSTICS}:${GH_COMPILER_TIME_TRACE}:${GH_DEVICE_SANITIZE}:${CMAKE_BUILD_TYPE}:${CMAKE_CXX_FLAGS}:${CMAKE_CXX_FLAGS_RELEASE}:${CMAKE_CXX_FLAGS_DEBUG}:${CMAKE_EXE_LINKER_FLAGS}:${CUDAToolkit_VERSION}")
  foreach(file IN ITEMS CMakeLists.txt include/ghb/instrumentation.hpp src/instrumentation.cpp bench/instrumentation.cpp)
    file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/${file}" digest)
    string(APPEND fingerprint ":${file}:${digest}")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/${file}")
  endforeach()
  foreach(file IN ITEMS "${GHB_COUNT_BUILD}/libgh.a" "${GHB_COUNT_BUILD}/libgh_policy.a" "${GH_ROOT}/include/gh/histogram.hpp" "${GH_ROOT}/support/common.hpp" "${GH_ROOT}/support/profiling.hpp")
    file(SHA256 "${file}" digest)
    string(APPEND fingerprint ":${digest}")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${file}")
  endforeach()
  string(SHA256 build_id "${fingerprint}")
  add_executable(ghb_instrumentation_bench bench/instrumentation.cpp)
  target_include_directories(ghb_instrumentation_bench PRIVATE "${GH_ROOT}/include" "${GH_ROOT}/support")
  target_compile_definitions(ghb_instrumentation_bench PRIVATE GHB_BUILD_ID="${build_id}")
  target_compile_options(ghb_instrumentation_bench PRIVATE -Wall -Wextra -Wpedantic)
  target_link_libraries(ghb_instrumentation_bench PRIVATE ghb_instrumentation ghb_existing_gh ghb_existing_gh_policy)
  if(GHB_BUILD_DIAGNOSTIC_VALIDATION)
    add_executable(ghb_validate_count profiling/validate_count.cpp)
    target_include_directories(ghb_validate_count PRIVATE "${GH_ROOT}/include")
    target_link_libraries(ghb_validate_count PRIVATE CUDA::cudart ghb_existing_gh ghb_existing_gh_policy)
  endif()
  set(probe_modes off timing)
  if(GHB_WITH_NVTX)
    list(APPEND probe_modes nvtx)
  endif()
  foreach(mode IN LISTS probe_modes)
    gh_add_gpu_test(NAME instrumentation_probe_${mode} COMMAND ghb_instrumentation_bench
      --n 131089 --bins 32768 --repetitions 3 --batch 2 --warmup-ms 0 --instrumentation ${mode} --launch graph)
    set_tests_properties(instrumentation_probe_${mode} PROPERTIES LABELS "gpu" RUN_SERIAL TRUE)
  endforeach()
endif()
